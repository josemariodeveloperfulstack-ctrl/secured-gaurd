Abra o DBeaver, pgAdmin ou qualquer cliente PostgreSQL e execute:

-----------------------------------------------------
1. VERIFICAR USUÁRIO
-----------------------------------------------------

SELECT 
    u.username,
    u.email,
    u.name,
    u.active,
    u.status,
    STRING_AGG(r.name, ', ') as roles,
    COUNT(r.id) as total_roles
FROM users u
LEFT JOIN user_roles ur ON u.id = ur.user_id
LEFT JOIN roles r ON ur.role_id = r.id
WHERE u.username = '05986003616' OR u.email LIKE '%05986003616%'
GROUP BY u.username, u.email, u.name, u.active, u.status;

-----------------------------------------------------
2. SE NÃO TIVER ROLES (total_roles = 0), EXECUTAR:
-----------------------------------------------------

-- Criar role COLABORADOR
INSERT INTO roles (id, name, description, created_at)
VALUES (gen_random_uuid(), 'COLABORADOR', 'Funcionário colaborador', NOW())
ON CONFLICT (name) DO NOTHING;

-- Associar role ao usuário
INSERT INTO user_roles (user_id, role_id)
SELECT u.id, r.id
FROM users u, roles r
WHERE u.username = '05986003616'
  AND r.name = 'COLABORADOR';

-----------------------------------------------------
3. VERIFICAR RESULTADO
-----------------------------------------------------

SELECT 
    u.username,
    u.email,
    r.name as role
FROM users u
JOIN user_roles ur ON u.id = ur.user_id
JOIN roles r ON ur.role_id = r.id
WHERE u.username = '05986003616';
